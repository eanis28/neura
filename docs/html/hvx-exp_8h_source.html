23static inline HVX_Vector hvx_vec_exp_f32(HVX_Vector in_vec) {
24 HVX_Vector z_qf32_v;
25 HVX_Vector x_v;
26 HVX_Vector x_qf32_v;
27 HVX_Vector y_v;
28 HVX_Vector k_v;
29 HVX_Vector f_v;
30 HVX_Vector epsilon_v;
31 HVX_Vector log2e = Q6_V_vsplat_R(EXP_LOG2E);
32 HVX_Vector logn2 = Q6_V_vsplat_R(EXP_LOGN2);
33 HVX_Vector E_const;
34 HVX_Vector zero_v = Q6_V_vzero();
35
36 // exp(x) is approximated as follows:
37 // f = floor(x/ln(2)) = floor(x*log2(e))
38 // epsilon = x - f*ln(2)
39 // exp(x) = exp(epsilon+f*ln(2))
40 // = exp(epsilon)*exp(f*ln(2))
41 // = exp(epsilon)*2^f
42 //
43 // Since epsilon is close to zero, it can be approximated with its Taylor series:
44 // exp(x) ~= 1+x+x^2/2!+x^3/3!+...+x^n/n!+...
45 // Preserving the first eight elements, we get:
46 // exp(x) ~= 1+x+e0*x^2+e1*x^3+e2*x^4+e3*x^5+e4*x^6+e5*x^7
47 // = 1+x+(E0+(E1+(E2+(E3+(E4+E5*x)*x)*x)*x)*x)*x^2
48
49 HVX_Vector temp_v = in_vec;
50
51 // Clamp inputs to (-88.0, 88.0) to avoid overflow/underflow
52 HVX_VectorPred pred_cap_right = Q6_Q_vcmp_gt_VsfVsf(in_vec, Q6_V_vsplat_R(EXP_RANGE_R));
53 HVX_VectorPred pred_cap_left = Q6_Q_vcmp_gt_VsfVsf(Q6_V_vsplat_R(EXP_RANGE_L), in_vec);
54
55 in_vec = Q6_V_vmux_QVV(pred_cap_right, Q6_V_vsplat_R(EXP_RANGE_R), temp_v);
56 in_vec = Q6_V_vmux_QVV(pred_cap_left, Q6_V_vsplat_R(EXP_RANGE_L), in_vec);
57
58 epsilon_v = Q6_Vqf32_vmpy_VsfVsf(log2e, in_vec);
59 epsilon_v = Q6_Vsf_equals_Vqf32(epsilon_v);
60
61 // f_v is the floating point result and k_v is the integer result
62 f_v = hvx_vec_floor_f32(epsilon_v);
63 k_v = hvx_vec_truncate_f32(f_v);
64
65 x_qf32_v = Q6_Vqf32_vadd_VsfVsf(in_vec, zero_v);
66
67 // x = x - f_v * logn2;
68 epsilon_v = Q6_Vqf32_vmpy_VsfVsf(f_v, logn2);
69 x_qf32_v = Q6_Vqf32_vsub_Vqf32Vqf32(x_qf32_v, epsilon_v);
70 // normalize before every QFloat's vmpy
71 x_qf32_v = Q6_Vqf32_vadd_Vqf32Vsf(x_qf32_v, zero_v);
72
73 x_v = Q6_Vsf_equals_Vqf32(x_qf32_v);
74
75 // z = x * x;
76 z_qf32_v = Q6_Vqf32_vmpy_Vqf32Vqf32(x_qf32_v, x_qf32_v);
77 z_qf32_v = Q6_Vqf32_vadd_Vqf32Vsf(z_qf32_v, zero_v);
78
79 // y = E4 + E5 * x;
80 E_const = Q6_V_vsplat_R(EXP_COEFF_5);
81 y_v = Q6_Vqf32_vmpy_VsfVsf(E_const, x_v);
82 E_const = Q6_V_vsplat_R(EXP_COEFF_4);
83 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, E_const);
84 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, zero_v);
85
86 // y = E3 + y * x;
87 E_const = Q6_V_vsplat_R(EXP_COEFF_3);
88 y_v = Q6_Vqf32_vmpy_Vqf32Vqf32(y_v, x_qf32_v);
89 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, E_const);
90 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, zero_v);
91
92 // y = E2 + y * x;
93 E_const = Q6_V_vsplat_R(EXP_COEFF_2);
94 y_v = Q6_Vqf32_vmpy_Vqf32Vqf32(y_v, x_qf32_v);
95 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, E_const);
96 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, zero_v);
97
98 // y = E1 + y * x;
99 E_const = Q6_V_vsplat_R(EXP_COEFF_1);
100 y_v = Q6_Vqf32_vmpy_Vqf32Vqf32(y_v, x_qf32_v);
101 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, E_const);
102 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, zero_v);
103
104 // y = E0 + y * x;
105 E_const = Q6_V_vsplat_R(EXP_COEFF_0);
106 y_v = Q6_Vqf32_vmpy_Vqf32Vqf32(y_v, x_qf32_v);
107 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, E_const);
108 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, zero_v);
109
110 // y = x + y * z;
111 y_v = Q6_Vqf32_vmpy_Vqf32Vqf32(y_v, z_qf32_v);
112 y_v = Q6_Vqf32_vadd_Vqf32Vqf32(y_v, x_qf32_v);
113 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, zero_v);
114
115 // y = y + 1.0;
116 y_v = Q6_Vqf32_vadd_Vqf32Vsf(y_v, Q6_V_vsplat_R(EXP_ONE));
117
118 // insert exponents
119 // y = ldexpf(y, k);
120 // y_v += k_v; // qf32
121 // modify exponent
122
123 y_v = Q6_Vsf_equals_Vqf32(y_v);
124
125 // add k_v to the exponent of y_v
126 HVX_Vector y_v_exponent = Q6_Vw_vasl_VwR(y_v, 1);
127
128 y_v_exponent = Q6_Vuw_vlsr_VuwR(y_v_exponent, IEEE_VSF_MANTLEN + 1);
129 y_v_exponent = Q6_Vw_vadd_VwVw(k_v, y_v_exponent);
130
131 // exponent cannot be negative; if overflow is detected, result is set to zero
132 HVX_VectorPred qy_v_negative_exponent = Q6_Q_vcmp_gt_VwVw(zero_v, y_v_exponent);
133
134 y_v = Q6_Vw_vaslacc_VwVwR(y_v, k_v, IEEE_VSF_MANTLEN);
135
136 y_v = Q6_V_vmux_QVV(qy_v_negative_exponent, zero_v, y_v);
137
138 return y_v;
139}

141static inline HVX_Vector hvx_vec_exp_f32_guard(HVX_Vector in_vec, HVX_Vector max_exp, HVX_Vector inf) {
142 const HVX_VectorPred pred0 = Q6_Q_vcmp_gt_VsfVsf(in_vec, max_exp);
143
144 HVX_Vector out = hvx_vec_exp_f32(in_vec);
145
146 return Q6_V_vmux_QVV(pred0, inf, out);
147}

149static inline void hvx_exp_f32(uint8_t * restrict dst, const uint8_t * restrict src, const int num_elems, bool negate) {
150 int left_over = num_elems & (VLEN_FP32 - 1);
151 int num_elems_whole = num_elems - left_over;
152
153 int unaligned_addr = 0;
154 int unaligned_loop = 0;
155 if ((0 == hex_is_aligned((void *) src, VLEN)) || (0 == hex_is_aligned((void *) dst, VLEN))) {
156 unaligned_addr = 1;
157 }
158 // assert((0 == unaligned_addr) || (0 == num_elems_whole));
159 if ((1 == unaligned_addr) && (num_elems_whole != 0)) {
160 unaligned_loop = 1;
161 }
162
163 HVX_Vector vec_out = Q6_V_vzero();
164
165 static const float kInf = INFINITY;
166 static const float kMaxExp = 88.7f;
167
168 const HVX_Vector max_exp = hvx_vec_splat_f32(kMaxExp);
169 const HVX_Vector inf = hvx_vec_splat_f32(kInf);
170
171 if (0 == unaligned_loop) {
172 HVX_Vector * p_vec_in1 = (HVX_Vector *) src;
173 HVX_Vector * p_vec_out = (HVX_Vector *) dst;
174
175 #pragma unroll(4)
176 for (int i = 0; i < num_elems_whole; i += VLEN_FP32) {
177 if (true == negate) {
178 HVX_Vector neg_vec_in = hvx_vec_neg_f32(*p_vec_in1++);
179 *p_vec_out++ = hvx_vec_exp_f32_guard(neg_vec_in, max_exp, inf);
180 } else {
181 *p_vec_out++ = hvx_vec_exp_f32_guard(*p_vec_in1++, max_exp, inf);
182 }
183 }
184 } else {
185 #pragma unroll(4)
186 for (int i = 0; i < num_elems_whole; i += VLEN_FP32) {
187 HVX_Vector in = *(HVX_UVector *) (src + i * SIZEOF_FP32);
188
189 if (true == negate) {
190 HVX_Vector neg_vec_in = hvx_vec_neg_f32(in);
191 *(HVX_UVector *) (dst + i * SIZEOF_FP32) = hvx_vec_exp_f32_guard(neg_vec_in, max_exp, inf);
192 } else {
193 *(HVX_UVector *) (dst + i * SIZEOF_FP32) = hvx_vec_exp_f32_guard(in, max_exp, inf);
194 }
195 }
196 }
197
198 if (left_over > 0) {
199 const float * srcf = (float *) src + num_elems_whole;
200 float * dstf = (float *) dst + num_elems_whole;
201
202 HVX_Vector in = *(HVX_UVector *) srcf;
203
204 if (true == negate) {
205 HVX_Vector neg_vec_in = hvx_vec_neg_f32(in);
206
207 vec_out = hvx_vec_exp_f32_guard(neg_vec_in, max_exp, inf);
208 } else {
209 vec_out = hvx_vec_exp_f32_guard(in, max_exp, inf);
210 }
211
212 hvx_vec_store_u((void *) dstf, left_over * SIZEOF_FP32, vec_out);
213 }
214}
