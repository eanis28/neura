88// =====================================================
89// OpenVINO Tensor Extra Types
90// =====================================================
91// These types are stored in tensor->extra by the OpenVINO backend buffer.
92// They allow:
93// 1. Pre-built ov::Constant nodes for weights (avoiding memcpy during graph construction)
94// 2. ov::Tensor wrappers for KV cache / compute tensors (for direct use with infer_request)
95
96// Base class for OpenVINO tensor extra data
97struct ggml_openvino_extra_base {
98 enum class Type { WEIGHT, QUANTIZED_WEIGHT, TENSOR };
99 Type type;
100 virtual ~ggml_openvino_extra_base() = default;
101protected:
102 explicit ggml_openvino_extra_base(Type t) : type(t) {}
103};
104
105// Extra data for F16/F32/BF16 weight tensors - stores the pre-built weight node
106struct ggml_openvino_weight_extra : public ggml_openvino_extra_base {
107 ov::Tensor weights; // The underlying weight data tensor
108 std::shared_ptr<ov::Node> weight_node; // Pre-built OpenVINO weight node
109
110 ggml_openvino_weight_extra(ov::Tensor w, std::shared_ptr<ov::Node> n) :
111 ggml_openvino_extra_base(Type::WEIGHT),
112 weights(std::move(w)),
113 weight_node(std::move(n)) {}
114};
115
116// Extra data for quantized weight tensors - stores extracted weights/scales/zp and weight node
117struct ggml_openvino_quantized_weight_extra : public ggml_openvino_extra_base {
118 ov::Tensor weights; // U4 or U8 extracted weights
119 ov::Tensor scales; // F16 scales
120 ov::Tensor zp; // U4 or U8 zero points (same type as weights)
121 std::shared_ptr<ov::Node> weight_node; // Pre-built OpenVINO weight subgraph
122
123 ggml_openvino_quantized_weight_extra(ov::Tensor w, ov::Tensor s, ov::Tensor z, std::shared_ptr<ov::Node> n) :
124 ggml_openvino_extra_base(Type::QUANTIZED_WEIGHT),
125 weights(std::move(w)),
126 scales(std::move(s)),
127 zp(std::move(z)),
128 weight_node(std::move(n)) {}
129};
130
131// Extra data for KV cache / compute tensors - stores ov::Tensor for infer_request
132struct ggml_openvino_tensor_extra : public ggml_openvino_extra_base {
133 std::shared_ptr<ov::Tensor> tensor; // For direct use with infer_request
134
135 explicit ggml_openvino_tensor_extra(std::shared_ptr<ov::Tensor> t)
136 : ggml_openvino_extra_base(Type::TENSOR), tensor(std::move(t)) {}
137};
