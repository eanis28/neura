135static inline bool dma_queue_push_single_1d(dma_queue * q, dma_ptr dptr, size_t size) {
136 if (((q->push_idx + 1) & q->idx_mask) == q->pop_idx) {
137 FARF(HIGH, "dma-push: queue full\n");
138 return false;
139 }
140
141 dma_descriptor_1d * desc = (dma_descriptor_1d *) &q->desc[q->push_idx];
142 desc->next = NULL;
143 desc->desc_size = 0; // 1D mode
144 desc->src_bypass = dma_src_l2_bypass_on;
145 desc->dst_bypass = dma_dst_l2_bypass_on;
146 desc->order = 1;
147 desc->done = 0;
148 desc->src = (void *) dptr.src;
149 desc->dst = (void *) dptr.dst;
150 desc->size = size;
151
152 q->dptr[q->push_idx] = dptr;
153
154 dmlink(q->tail, desc);
155 q->tail = (dma_descriptor_2d *) desc;
156
157 // FARF(ERROR, "dma-push: i %u row-size %u nrows %d dst %p src %p\n", q->push_idx, row_size, nrows, dptr.dst, dptr.src);
158 q->push_idx = (q->push_idx + 1) & q->idx_mask;
159 return true;
160}

208static inline dma_ptr dma_queue_pop(dma_queue * q) {
209 dma_ptr dptr = { NULL };
210
211 if (q->push_idx == q->pop_idx) {
212 return dptr;
213 }
214
215 dma_descriptor_2d * desc = &q->desc[q->pop_idx];
216
217 // Wait for desc to complete
218 while (1) {
219 dmpoll();
220 if (desc->done) {
221 break;
222 }
223 // FARF(ERROR, "dma-pop: waiting for DMA : %u\n", q->pop_idx);
224 }
225
226 dptr = q->dptr[q->pop_idx];
227
228 // FARF(ERROR, "dma-pop: i %u dst %p src %p\n", q->pop_idx, dptr.dst, dptr.src);
229 q->pop_idx = (q->pop_idx + 1) & q->idx_mask;
230 return dptr;
231}

261// Overflow-safe DMA push: all 2d descriptor fields (row_size, nrows, src_stride, dst_stride) are 16-bit, max 65535.
262// This version transparently handles values that exceed the 16-bit limit and submits chained DMA transtions.
263
264#define DMA_MAX_FIELD_VAL 65535u
265
266static inline bool dma_queue_push(dma_queue *q, dma_ptr dptr, size_t dst_stride, size_t src_stride, size_t row_size, size_t nrows) {
267 // Fast path: everything fits in 16 bits
268 if (nrows == 0 || __builtin_expect(
269 row_size <= DMA_MAX_FIELD_VAL &&
270 nrows <= DMA_MAX_FIELD_VAL &&
271 src_stride <= DMA_MAX_FIELD_VAL &&
272 dst_stride <= DMA_MAX_FIELD_VAL, 1)) {
273 return dma_queue_push_single_2d(q, dptr, dst_stride, src_stride, row_size, nrows);
274 }
275
276 // Contiguous block
277 // Use 1d DMA mode which supports sizes up to 24-bits (16MB)
278 if (nrows == 1 || (row_size == src_stride && row_size == dst_stride)) {
279 size_t total = row_size * nrows;
280 return dma_queue_push_single_1d(q, dptr, total);
281 }
282
283 // Stride overflow — fall back to row-by-row.
284 {
285 const uint8_t *src = (const uint8_t *) dptr.src;
286 uint8_t *dst = (uint8_t *) dptr.dst;
287 for (size_t r = 0; r < nrows; ++r) {
288 dma_ptr p = dma_make_ptr(dst + r * dst_stride, src + r * src_stride);
289 if (!dma_queue_push_single_1d(q, p, row_size))
290 return false;
291 if (r + 1 < nrows)
292 dma_queue_pop(q);
293 }
294 return true;
295 }
296}
