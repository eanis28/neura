932static bool is_matmul_weight(const ggml_tensor * tensor) {
933 std::string name = ggml_get_name(tensor);
934 static const std::unordered_set<std::string> weight_suffixes{ "output.weight", "attn_q.weight",
935 "attn_k.weight", "attn_v.weight",
936 "attn_output.weight", "ffn_gate.weight",
937 "ffn_up.weight", "ffn_down.weight" };
938
939 for (const auto & suffix : weight_suffixes) {
940 if (name.find(suffix) != std::string::npos) {
941 return true;
942 }
943 }
944 return false;
945}
