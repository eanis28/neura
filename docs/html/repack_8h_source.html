13template <int K> constexpr int QK_0() {
14 if constexpr (K == 4) {
15 return QK4_0;

18 return QK8_0;
19 }

23template <int K, int N> struct block {
24 ggml_half d[N]; // deltas for N qK_0 blocks
25 int8_t qs[(QK_0<K>() * N * K) / 8]; // quants for N qK_0 blocks
26};
