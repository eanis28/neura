424// This is exposed because it's necessary in supports_op
425inline size_t ggml_webgpu_flash_attn_wg_mem_bytes(uint32_t q_tile,
426 uint32_t kv_tile,
427 uint32_t head_dim_qk,
428 uint32_t head_dim_v,
429 bool has_mask,
430 bool kv_direct) {
431 const uint32_t max_head_dim = std::max(head_dim_qk, head_dim_v);
432 size_t f16_elems = 0;
433 size_t f32_elems = 0;
434 f16_elems += q_tile * head_dim_qk; // q_shmem
435 if (!kv_direct) {
436 f16_elems += kv_tile * max_head_dim; // kv_shmem
437 }
438 f16_elems += q_tile * head_dim_v; // o_shmem
439 if (has_mask) {
440 f16_elems += q_tile * kv_tile; // mask_shmem
441 }
442 f16_elems += q_tile * kv_tile; // inter_shmem
443 f32_elems += q_tile; // row_max_shmem
444 f32_elems += q_tile; // exp_sum_shmem
445 return f16_elems * GGML_WEBGPU_F16_SIZE_BYTES + f32_elems * GGML_WEBGPU_F32_SIZE_BYTES;
446}
